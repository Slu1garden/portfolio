Токены

Единый источник значений для новых экранов.

:root{
  --p: #746EAB;  /* Primary */
  --pd: #585289;  /* Primary dark */
  --pl: #C1BFD9;  /* Lilac */
  --pm: #9A96C2;  /* Lilac mid */
  --y: #F5FF64;  /* Accent yellow */
  --s: #F4F5F8;  /* Surface */
  --w: #FFFFFF;  /* White */
  --sk: #E2E6F1;  /* Skeleton */
  --tx: #2B2A34;  /* Ink */
  --tx2: #7B7B82;  /* Ink 2 */
  --g: #BDF28F;  /* Chart green */
  --v: #AD94FF;  /* Chart violet */
  --b: #9DD3FC;  /* Chart blue */
  --r-pill: 999px; --r-head: 48px; --r-modal: 24px; --r-card: 20px; --r-tile: 16px;
  --sp: 4px 8px 12px 16px 24px 32px;  /* шаг сетки */
  --font: Inter, system-ui, sans-serif;
}
